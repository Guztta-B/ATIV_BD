CREATE DATABASE SALA_DE_BELEZA;

--------------------------------------------------

CRIAR TABELAS
cliente

CREATE TABLE cliente (
id_cliente INT PRIMARY KEY AUTO_INCREMENT,
nome VARCHAR(50),
telefone INT(10)
 );

---------------------------------------------------
serviços

CREATE TABLE servicos (
    id_servico INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(50) NOT NULL,
    categoria VARCHAR(30),
    preco DECIMAL(8,2),
    duracao_minutos INT
);

CREATE TABLE funcionarios (
ID INT PRIMARY KEY AUTO_INCREMENT,
nome VARCHAR(45),
cpf INT(15),
telefone INT(10),
cargo VARCHAR(30)
);

------------------------------------------------------------------------------

AGENDAMENTO

CREATE TABLE agendamento (
    id_agendamento INT PRIMARY KEY AUTO_INCREMENT,
    cliente VARCHAR(50) NOT NULL,
    data_hora DATETIME NOT NULL,
    servico VARCHAR(50) NOT NULL
);

------------------------------------------------------------------------------


PAGAMENTO


CREATE TABLE pagamentos ( id_pagamento INT PRIMARY KEY AUTO_INCREMENT, numero_atendimento INT(10), nome_cliente VARCHAR(50), status_pagamento VARCHAR(10), metodo_de_pagamento VARCHAR(10), valor DECIMAL(10, 2), data_inicial DATETIME NOT NULL, data_final DATETIME NOT NULL );
